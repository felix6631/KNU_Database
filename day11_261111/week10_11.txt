  `nickname` char(100) DEFAULT NULL,
  `course` char(100) DEFAULT NULL
);

  `nickname` varchar(30) DEFAULT NULL,
  `food` varchar(30) DEFAULT NULL
);

  `nickname` varchar(30) DEFAULT NULL,
  `friend_nickname` varchar(30) DEFAULT NULL
);

  `nickname` varchar(30) DEFAULT NULL,
  `lang` varchar(30) DEFAULT NULL
);

  `nickname` varchar(30) DEFAULT NULL,
  `section` varchar(15) DEFAULT NULL
);


CREATE TABLE `career_path` (
  `nickname` varchar(30) NOT NULL,
  `direction` varchar(30) DEFAULT NULL);

  `nickname` varchar(30) NOT NULL,
  `sauce` varchar(30) DEFAULT NULL,
  `mint` varchar(30) DEFAULT NULL
);

 
CREATE TABLE `mbti` (
  `nickname` char(100) NOT NULL,
  `gender` char(10) DEFAULT NULL,
  `mbti_type` char(4) DEFAULT NULL
);
